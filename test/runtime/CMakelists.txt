# 精确测试输入部署到独立资产根；测试界面复用同一份生产字体及许可。
set(RUNTIME_TEST_ASSET_DIR "${CMAKE_BINARY_DIR}/test-fixtures/assets")
set(RUNTIME_TEST_FIXTURE_OUTPUTS)
foreach(FONT_FILE IN ITEMS NotoSansSC-Regular.otf OFL.txt)
    set(FONT_OUTPUT "${RUNTIME_TEST_ASSET_DIR}/fonts/${FONT_FILE}")
    add_custom_command(OUTPUT "${FONT_OUTPUT}"
        COMMAND "${CMAKE_COMMAND}" -E make_directory "${RUNTIME_TEST_ASSET_DIR}/fonts"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${ASSETS_DIR}/fonts/${FONT_FILE}" "${FONT_OUTPUT}"
        DEPENDS "${ASSETS_DIR}/fonts/${FONT_FILE}"
        COMMENT "部署测试界面字体及许可 ${FONT_FILE}"
        VERBATIM)
    list(APPEND RUNTIME_TEST_FIXTURE_OUTPUTS "${FONT_OUTPUT}")
endforeach()

# 收集并归并全部无窗口纯 CPU 测试为单一 runtime_cpu_test 目标。
file(GLOB TEST_SOURCES CONFIGURE_DEPENDS "*.cpp")
set(RUNTIME_GPU_WINDOW_TESTS
    "${CMAKE_CURRENT_SOURCE_DIR}/imgui_integration_test.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/model_preparation_test.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/render_input_test.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/vulkan_review_test.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/window_test.cpp"
)
set(RUNTIME_CPU_SOURCES ${TEST_SOURCES})
list(REMOVE_ITEM RUNTIME_CPU_SOURCES ${RUNTIME_GPU_WINDOW_TESTS})

add_executable(runtime_cpu_test ${RUNTIME_CPU_SOURCES})
target_link_libraries(runtime_cpu_test PRIVATE GTest::gtest_main GTest::gmock_main IncubatorRuntime assimp::assimp)
target_include_directories(runtime_cpu_test PRIVATE ${CMAKE_SOURCE_DIR}/src ${CMAKE_SOURCE_DIR}/test)
target_include_directories(runtime_cpu_test SYSTEM PRIVATE "${THIRD_PARTY_DIR}/assimp/contrib/rapidjson/include")
target_compile_definitions(runtime_cpu_test PRIVATE
    SCENE_FIXTURE_DIR="${RUNTIME_TEST_ASSET_DIR}/scenes"
    MODEL_BUILD_ASSET_DIR="${RUNTIME_TEST_ASSET_DIR}"
    MODEL_DEMO_ASSET_DIR="${CMAKE_BINARY_DIR}/assets"
    TEXTURE_TEST_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures/textures"
    TEXTURE_REFERENCE_FILE="${RUNTIME_TEST_ASSET_DIR}/textures/reference_rgba.png"
    TEXTURE_BUILD_ASSET_DIR="${RUNTIME_TEST_ASSET_DIR}"
)
add_dependencies(runtime_cpu_test runtime_test_fixtures runtime_test_shaders incubator_models)
set_target_properties(runtime_cpu_test PROPERTIES FOLDER "UnitTest/runtime")
gtest_discover_tests(runtime_cpu_test PROPERTIES LABELS "cpu\\;runtime")

# 需要窗口或 Vulkan GPU 的独立测试目标保持按工作目录与串行属性注册。
foreach(TEST_SOURCE IN LISTS RUNTIME_GPU_WINDOW_TESTS)
    get_filename_component(TEST_NAME ${TEST_SOURCE} NAME_WE)
    add_executable(${TEST_NAME} ${TEST_SOURCE})
    target_link_libraries(${TEST_NAME} PRIVATE GTest::gtest_main GTest::gmock_main IncubatorRuntime)
    target_include_directories(${TEST_NAME} PRIVATE ${CMAKE_SOURCE_DIR}/src ${CMAKE_SOURCE_DIR}/test)
    set_target_properties(${TEST_NAME} PROPERTIES FOLDER "UnitTest/runtime")

    if(TEST_NAME STREQUAL "render_input_test")
        target_link_libraries(${TEST_NAME} PRIVATE vulkan)
        gtest_discover_tests(${TEST_NAME} PROPERTIES LABELS "cpu\\;runtime")
    elseif(TEST_NAME STREQUAL "window_test")
        gtest_discover_tests(${TEST_NAME} PROPERTIES LABELS "window\\;runtime")
    else()
        target_link_libraries(${TEST_NAME} PRIVATE vulkan imgui)
        add_dependencies(${TEST_NAME} runtime_test_fixtures runtime_test_shaders)
        target_compile_definitions(${TEST_NAME} PRIVATE RUNTIME_TEST_ASSET_DIR="${RUNTIME_TEST_ASSET_DIR}")
        if(TEST_NAME STREQUAL "model_preparation_test")
            add_dependencies(${TEST_NAME} incubator_models)
            gtest_discover_tests(${TEST_NAME}
                WORKING_DIRECTORY "${CMAKE_BINARY_DIR}"
                PROPERTIES RUN_SERIAL TRUE TIMEOUT 120 LABELS "gpu\\;runtime")
        else()
            gtest_discover_tests(${TEST_NAME}
                WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/test-fixtures"
                PROPERTIES RUN_SERIAL TRUE TIMEOUT 120 LABELS "gpu\\;runtime")
        endif()
    endif()
endforeach()
