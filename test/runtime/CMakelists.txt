# test/runtime CMakelists.txt

# 设置测试源文件列表
file(GLOB TEST_SOURCES CONFIGURE_DEPENDS "*.cpp")

# 创建测试源文件列表
foreach(TEST_SOURCE ${TEST_SOURCES})
    get_filename_component(TEST_NAME ${TEST_SOURCE} NAME_WE)

    add_executable(${TEST_NAME} ${TEST_SOURCE})

    target_link_libraries(
        ${TEST_NAME} PRIVATE
        GTest::gtest_main
        GTest::gmock_main
        IncubatorRuntime
    )

    target_include_directories(
        ${TEST_NAME} PRIVATE
        ${CMAKE_SOURCE_DIR}/src
    )

    # 设置测试可执行文件的目录
    set_target_properties(${TEST_NAME} PROPERTIES FOLDER "UnitTest/runtime")

    # 注册测试到 CMake
    include(GoogleTest)
    if(TEST_NAME STREQUAL "vulkan_review_test")
        target_include_directories(${TEST_NAME} PRIVATE "${CMAKE_SOURCE_DIR}")
        target_link_libraries(${TEST_NAME} PRIVATE vulkan imgui)
        add_dependencies(${TEST_NAME} demo_shaders demo_textures)
        gtest_discover_tests(${TEST_NAME}
            WORKING_DIRECTORY "${CMAKE_BINARY_DIR}"
            PROPERTIES RUN_SERIAL TRUE TIMEOUT 120)
    else()
        if(TEST_NAME STREQUAL "texture_loader_test")
            target_compile_definitions(${TEST_NAME} PRIVATE
                TEXTURE_TEST_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures/textures"
                TEXTURE_REFERENCE_FILE="${ASSETS_DIR}/textures/reference_rgba.png"
                TEXTURE_BUILD_ASSET_DIR="${CMAKE_BINARY_DIR}/assets")
            add_dependencies(${TEST_NAME} demo_textures demo_shaders)
        endif()
        gtest_discover_tests(${TEST_NAME})
    endif()
endforeach()
