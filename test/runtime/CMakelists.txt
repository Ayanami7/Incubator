# Deploy exact test inputs under a test-only asset root. Demo builds never receive fixtures.
set(RUNTIME_TEST_ASSET_DIR "${CMAKE_BINARY_DIR}/test-fixtures/assets")
set(RUNTIME_TEST_FIXTURE_OUTPUTS)
foreach(MODEL_FILE IN ITEMS
    minimal_gltf/model.gltf
    minimal_gltf/geometry/mesh.bin
    minimal_obj/model.obj
    minimal_obj/materials/model.mtl)
    set(MODEL_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/models/${MODEL_FILE}")
    set(MODEL_OUTPUT "${RUNTIME_TEST_ASSET_DIR}/models/${MODEL_FILE}")
    get_filename_component(MODEL_OUTPUT_DIRECTORY "${MODEL_OUTPUT}" DIRECTORY)
    add_custom_command(
        OUTPUT "${MODEL_OUTPUT}"
        COMMAND "${CMAKE_COMMAND}" -E make_directory "${MODEL_OUTPUT_DIRECTORY}"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${MODEL_SOURCE}" "${MODEL_OUTPUT}"
        DEPENDS "${MODEL_SOURCE}"
        COMMENT "Deploying model test fixture ${MODEL_FILE}"
        VERBATIM
    )
    list(APPEND RUNTIME_TEST_FIXTURE_OUTPUTS "${MODEL_OUTPUT}")
endforeach()
foreach(TEXTURE_FILE IN ITEMS reference_rgba.png direction_rgba.png)
    set(TEXTURE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/textures/${TEXTURE_FILE}")
    set(TEXTURE_OUTPUT "${RUNTIME_TEST_ASSET_DIR}/textures/${TEXTURE_FILE}")
    add_custom_command(
        OUTPUT "${TEXTURE_OUTPUT}"
        COMMAND "${CMAKE_COMMAND}" -E make_directory "${RUNTIME_TEST_ASSET_DIR}/textures"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${TEXTURE_SOURCE}" "${TEXTURE_OUTPUT}"
        DEPENDS "${TEXTURE_SOURCE}"
        COMMENT "Deploying texture test fixture ${TEXTURE_FILE}"
        VERBATIM
    )
    list(APPEND RUNTIME_TEST_FIXTURE_OUTPUTS "${TEXTURE_OUTPUT}")
endforeach()
add_custom_target(runtime_test_fixtures DEPENDS ${RUNTIME_TEST_FIXTURE_OUTPUTS})

set(RUNTIME_TEST_SHADER_OUTPUTS)
foreach(SHADER_FILE IN ITEMS cube.vert.spv cube.frag.spv)
    set(SHADER_SOURCE "${CMAKE_BINARY_DIR}/assets/shaders/${SHADER_FILE}")
    set(SHADER_OUTPUT "${RUNTIME_TEST_ASSET_DIR}/shaders/${SHADER_FILE}")
    add_custom_command(
        OUTPUT "${SHADER_OUTPUT}"
        COMMAND "${CMAKE_COMMAND}" -E make_directory "${RUNTIME_TEST_ASSET_DIR}/shaders"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${SHADER_SOURCE}" "${SHADER_OUTPUT}"
        DEPENDS "${SHADER_SOURCE}"
        COMMENT "Deploying production shader fixture ${SHADER_FILE}"
        VERBATIM
    )
    list(APPEND RUNTIME_TEST_SHADER_OUTPUTS "${SHADER_OUTPUT}")
endforeach()
add_custom_target(runtime_test_shaders DEPENDS ${RUNTIME_TEST_SHADER_OUTPUTS})
add_dependencies(runtime_test_shaders demo_shaders)

# Create test source file list.
file(GLOB TEST_SOURCES CONFIGURE_DEPENDS "*.cpp")

foreach(TEST_SOURCE ${TEST_SOURCES})
    get_filename_component(TEST_NAME ${TEST_SOURCE} NAME_WE)

    add_executable(${TEST_NAME} ${TEST_SOURCE})

    target_link_libraries(
        ${TEST_NAME} PRIVATE
        GTest::gtest_main
        GTest::gmock_main
        IncubatorRuntime
    )

    target_include_directories(
        ${TEST_NAME} PRIVATE
        ${CMAKE_SOURCE_DIR}/src
    )

    set_target_properties(${TEST_NAME} PROPERTIES FOLDER "UnitTest/runtime")

    include(GoogleTest)
    if(TEST_NAME STREQUAL "vulkan_review_test" OR TEST_NAME STREQUAL "model_preparation_test" OR
       TEST_NAME STREQUAL "application_test")
        target_include_directories(${TEST_NAME} PRIVATE "${CMAKE_SOURCE_DIR}")
        target_link_libraries(${TEST_NAME} PRIVATE vulkan imgui)
        add_dependencies(${TEST_NAME} runtime_test_fixtures runtime_test_shaders)
        if(TEST_NAME STREQUAL "model_preparation_test" OR TEST_NAME STREQUAL "application_test")
            target_compile_definitions(${TEST_NAME} PRIVATE RUNTIME_TEST_ASSET_DIR="${RUNTIME_TEST_ASSET_DIR}")
            add_dependencies(${TEST_NAME} demo_models)
        endif()
        if(TEST_NAME STREQUAL "vulkan_review_test")
            gtest_discover_tests(${TEST_NAME}
                WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/test-fixtures"
                PROPERTIES RUN_SERIAL TRUE TIMEOUT 120)
        else()
            gtest_discover_tests(${TEST_NAME}
                WORKING_DIRECTORY "${CMAKE_BINARY_DIR}"
                PROPERTIES RUN_SERIAL TRUE TIMEOUT 120)
        endif()
    else()
        if(TEST_NAME STREQUAL "assimp_fileio_test" OR TEST_NAME STREQUAL "model_loader_test")
            target_link_libraries(${TEST_NAME} PRIVATE assimp::assimp)
        endif()
        if(TEST_NAME STREQUAL "model_loader_test")
            target_include_directories(${TEST_NAME} SYSTEM PRIVATE "${THIRD_PARTY_DIR}/assimp/contrib/rapidjson/include")
            target_compile_definitions(${TEST_NAME} PRIVATE
                MODEL_BUILD_ASSET_DIR="${RUNTIME_TEST_ASSET_DIR}"
                MODEL_DEMO_ASSET_DIR="${CMAKE_BINARY_DIR}/assets")
            add_dependencies(${TEST_NAME} runtime_test_fixtures demo_models)
        endif()
        if(TEST_NAME STREQUAL "texture_loader_test")
            target_compile_definitions(${TEST_NAME} PRIVATE
                TEXTURE_TEST_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures/textures"
                TEXTURE_REFERENCE_FILE="${RUNTIME_TEST_ASSET_DIR}/textures/reference_rgba.png"
                TEXTURE_BUILD_ASSET_DIR="${RUNTIME_TEST_ASSET_DIR}")
            add_dependencies(${TEST_NAME} runtime_test_fixtures runtime_test_shaders)
        endif()
        gtest_discover_tests(${TEST_NAME})
    endif()
endforeach()
